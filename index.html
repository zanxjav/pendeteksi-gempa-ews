Copy kode ini dan upload menggunakan Arduino IDE ke board ESP32 Anda. (File lengkap tersedia di folder firmware/esp32_disaster_station.ino):


/*
 * GeoShield EWS - ESP32 Disaster Station Firmware
 * Sensor: MPU6050 (Gempa), HC-SR04 (Banjir), FC-37 (Hujan), Analog TDS
 */
#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <MPU6050.h>

const char* ssid = "NAMA_WIFI_ANDA";
const char* password = "PASSWORD_WIFI";
const char* serverUrl = "http://YOUR_SERVER_IP_OR_DOMAIN/api/telemetry";

MPU6050 mpu;
#define TRIG_PIN 5
#define ECHO_PIN 18
#define RAIN_PIN 34
#define TDS_PIN  35
#define BUZZER_PIN 4

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);
  mpu.initialize();
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(RAIN_PIN, INPUT);
  pinMode(TDS_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nWiFi Terkoneksi!");
}

void loop() {
  // 1. Baca Akselerasi MPU6050 (Gempa)
  int16_t ax, ay, az;
  mpu.getAcceleration(&ax, &ay, &az);
  float pga_g = sqrt(ax*ax + ay*ay + az*az) / 16384.0;
  
  // 2. Baca Muka Air HC-SR04 (Banjir)
  digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH);
  float water_cm = duration * 0.034 / 2;
  
  // 3. Baca Sensor Hujan & TDS
  int rainRaw = analogRead(RAIN_PIN);
  int tdsRaw = analogRead(TDS_PIN);
  float tdsPpm = (tdsRaw / 4095.0) * 1000.0;
  
  // Kirim JSON ke Web Dashboard
  if(WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverUrl);
    http.addHeader("Content-Type", "application/json");
    String payload = String("{\"station_id\":\"ST-01\",\"pga\":") + pga_g + 
                     ",\"water_level\":" + water_cm + 
                     ",\"rain_raw\":" + rainRaw + 
                     ",\"tds_ppm\":" + tdsPpm + "}";
    int httpResponseCode = http.POST(payload);
    http.end();
  }
  delay(100);
}
